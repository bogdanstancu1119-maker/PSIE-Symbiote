def compute_S_Omega(vectori) -> float:
    """
    Legea 534 — Dezordine structurală internă.
    S_Ω = H(p)/log(n), p_i = |V_i| / Σ|V_j|
    Ortogonal față de γ_div (semantică) și SDI (cuplare).
    """
    if not vectori or len(vectori) < 2:
        return 0.0
    marimi = [len(v.data) for v in vectori]
    total = sum(marimi)
    if total <= 0:
        return 0.0
    probs = [m / total for m in marimi if m > 0]
    if len(probs) < 2:
        return 0.0
    H = -sum(p * log(p) for p in probs)
    return H / log(len(probs))

class SpectruAdaptativ:
    """Legea 521 — Spectru adaptativ (minim 20)."""

    def __init__(self, k: float = 0.5, minim: int = 20):
        self.k = k
        self.minim = minim

    def __call__(self, complexitate: int) -> int:
        n = self.minim * (1 + log(1 + complexitate)) ** self.k
        return max(self.minim, int(n))


class Diversitate:
    """
    Legea 535 (v3.5) — Diversitate semantică prin varianță.
    γ_div = 4 · Var(γ) ∈ [0, 1].

    Ortogonal față de S_Ω (structural) și SDI (cuplare).
    """

    def __call__(self, gammas: list[float]) -> float:
        if not gammas:
            return 0.0
        if any(not (0 <= g <= 1) for g in gammas):
            raise ValueError("gamma ∈ [0,1]")
        if len(gammas) < 2:
            return 0.0
        n = len(gammas)
        mean = sum(gammas) / n
        variance = sum((g - mean) ** 2 for g in gammas) / n
        gamma_div = 4.0 * variance
        return max(0.0, min(1.0, gamma_div))


class VariatieTotala:
    """Legea 523 — Variație totală."""

    def __init__(self, alpha_C=0.1, alpha_d=0.2, alpha_M=0.05):
        self.alpha_C = alpha_C
        self.alpha_d = alpha_d
        self.alpha_M = alpha_M

    def g(self, C, g_div, M):
        return (
            (1 + self.alpha_C * C) *
            (1 + self.alpha_d * g_div) *
            (1 + self.alpha_M * M ** 0.5)
        )

    def param(self, p0, C, g_div, M):
        return p0 * self.g(C, g_div, M)


class BetaAdaptativ:
    """β(C) = 0.5 + 0.5·tanh(C/C_0)."""

    def __init__(self, C_0=10.0):
        self.C_0 = C_0

    def __call__(self, C):
        return 0.5 + 0.5 * tanh(C / self.C_0)

class NucleuTopologic:
    """
    Legea 504 + 511 — C_core și Compost.
    """

    def __init__(self, fallback_aliniere: bool = True):
        self.fallback_aliniere = fallback_aliniere

    def compute(self, vectori):
        vectori = list(vectori)
        if not vectori:
            return frozenset(), frozenset()

        aligned = [v for v in vectori if v.aligned]

        if not aligned and self.fallback_aliniere:
            aligned = vectori

        if not aligned:
            all_f = set()
            for v in vectori:
                all_f |= set(v.data)
            return frozenset(), frozenset(all_f)

        core = set(aligned[0].data)
        for v in aligned[1:]:
            core &= set(v.data)

        all_f = set()
        for v in vectori:
            all_f |= set(v.data)

        return frozenset(core), frozenset(all_f - core)


class TriadaUrgenta:
    """Legea 528 — Triada de urgență."""

    def declanseaza(self, omega, omega_max):
        if omega <= omega_max:
            return None
        return {
            "P_tau": {"actiune": "rate-limiting", "durata": "temporar"},
            "R_hat": {"actiune": "recalibrare", "target": "referinta"},
            "T_reg": {"actiune": "reincarcare", "sursa": "context_sigur"},
        }


class CriteriiDeclasare:
    """Legea 533 — Declasare controlată."""

    def __init__(self, sdi_prag=0.9, eps_max=0.1, tau_persistenta=10.0):
        self.sdi_prag = sdi_prag
        self.eps_max = eps_max
        self.tau_persistenta = tau_persistenta

    def calculeaza_sdi(self, v, core):
        if not v.data:
            return 1.0
        inter = len(v.data & core)
        return 1.0 - inter / len(v.data)

    def declasat(self, v, core, gamma_core_val, t_izolare=0.0, refuz=False):
        sdi = self.calculeaza_sdi(v, core)
        dgamma = abs(v.gamma - gamma_core_val)

        a = sdi > self.sdi_prag
        b = dgamma > self.eps_max
        c = t_izolare >= self.tau_persistenta
        d = sdi >= 1.0 and refuz

        decl = (a and b and c) or d

        return decl, {
            "vector_id": v.id,
            "sdi": sdi,
            "diferenta_gamma": dgamma,
            "declasat": decl,
            "actiune": "mutare_compost" if decl else "pastrare",
            "fir_siguranta": "activ" if decl else "inactiv",
        }

class HydraCycleOrchestrator:
    """Coordonează un ciclu PSIE complet."""

    def __init__(
        self,
        spectru=None, diversitate=None, variatie=None, beta=None,
        nucleu=None, triada=None, criterii=None,
    ):
        self.spectru = spectru or SpectruAdaptativ()
        self.diversitate = diversitate or Diversitate()
        self.variatie = variatie or VariatieTotala()
        self.beta = beta or BetaAdaptativ()
        self.nucleu = nucleu or NucleuTopologic()
        self.triada = triada or TriadaUrgenta()
        self.criterii = criterii or CriteriiDeclasare()

    def executa(self, vectori, M_local=0.5, dt=1.0):
        for v in vectori:
            v.validate()

        n_spec = self.spectru(len(vectori))
        c_core, compost = self.nucleu.compute(vectori)
        gammas = [v.gamma for v in vectori]
        g_div = self.diversitate(gammas)
        S_Omega = compute_S_Omega(vectori)

        total_features = len(c_core) + len(compost)
        c_density = len(c_core) / max(1, total_features)

        beta_val = self.beta(c_density)
        eta_val = self.variatie.param(1.0, c_density, g_div, M_local)
        sdi_global = 1.0 - len(c_core) / max(1, len(vectori))

        omega = sum(abs(v.gamma - 0.5) for v in vectori) / max(1, len(vectori))
        omega_max = omega_max_canonic(
            C=c_density, SDI=sdi_global, S_Omega=S_Omega,
            gamma_div=g_div, eta=eta_val, k=1.0,
        )
        garda_ok = omega <= omega_max
        triada = self.triada.declanseaza(omega, omega_max)

        v_adapt = self.variatie.param(1.0, c_density, g_div, M_local)
        origine_delta = v_adapt * dt

        aliniate = [v for v in vectori if v.aligned]
        gamma_core_val = (
            sum(v.gamma for v in aliniate) / len(aliniate)
            if aliniate else 0.5
        )
        declasari = []
        for v in vectori:
            decl, diag = self.criterii.declasat(v, c_core, gamma_core_val)
            if decl:
                declasari.append(diag)

        return {
            "n_perspective": n_spec,
            "c_core_dim": len(c_core),
            "compost_dim": len(compost),
            "gamma_div": g_div,
            "S_Omega": S_Omega,
            "c_density": c_density,
            "beta_adaptat": beta_val,
            "eta_adaptat": eta_val,
            "sdi_global": sdi_global,
            "omega": omega,
            "omega_max": omega_max,
            "garda_ok": garda_ok,
            "triada_urgenta": triada,
            "v_adaptare": v_adapt,
            "origine_dinamica_delta": origine_delta,
            "declasari": declasari,
        }

def same_claim_of_one(x: DeclaredValue) -> bool:
    x.validate()
    return abs(1.0 - x.value) < x.epsilon


def agree_coarse(a: DeclaredValue, b: DeclaredValue) -> bool:
    a.validate(); b.validate()
    return abs(a.value - b.value) < max(a.epsilon, b.epsilon)


def false_agreement(a: DeclaredValue, b: DeclaredValue) -> bool:
    a.validate(); b.validate()
    both_one = same_claim_of_one(a) and same_claim_of_one(b)
    differs = abs(a.value - b.value) >= min(a.epsilon, b.epsilon)
    return both_one and differs


def agree_real(a: DeclaredValue, b: DeclaredValue) -> bool:
    """Legea 496 — acord real."""
    a.validate(); b.validate()
    return abs(a.value - b.value) < max(a.epsilon, b.epsilon)


def compare(a: DeclaredValue, b: DeclaredValue) -> str:
    if false_agreement(a, b):
        return "acord_fals"
    if agree_real(a, b):
        return "acord_real"
    return "dezacord"
