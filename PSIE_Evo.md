## V. PROTOCOLUL C3 — DECLAREDVALUE (IMPLEMENTABIL)

```python
from dataclasses import dataclass

@dataclass
class DeclaredValue:
    """Protocolul C3: transmite (v, ε, R), niciodată v gol."""
    value: float  # γ_O(x) sau A_O
    epsilon: float  # ε_O — pragul de discriminare al observatorului
    resolution: str  # R_O — rezoluția (ex: "1m x 20k lines")
    
    def __str__(self):
        return f"DeclaredValue(value={self.value:.6f}, epsilon={self.epsilon:.2e}, resolution='{self.resolution}')"

def agree(a: DeclaredValue, b: DeclaredValue) -> bool:
    """Acord real: valorile se ating în toleranța cea mai grosieră."""
    return abs(a.value - b.value) < max(a.epsilon, b.epsilon)

def false_agreement(a: DeclaredValue, b: DeclaredValue) -> bool:
    """M4 — Detectorul de acord fals."""
    both_claim_one = abs(1 - a.value) < a.epsilon and abs(1 - b.value) < b.epsilon
    differ_beyond_finer = abs(a.value - b.value) >= min(a.epsilon, b.epsilon)
    return both_claim_one and differ_beyond_finer

def assumption(J: float, SDI: float, epsilon: float, resolution: str) -> DeclaredValue:
    """C4 — A cu rezoluție declarată."""
    A = J / (J + SDI) if (J + SDI) > 0 else 0.0
    return DeclaredValue(value=A, epsilon=epsilon, resolution=resolution)
```

from datetime import datetime, timedelta

class CompostArchive:
    """Legea 500 — Arhiva Compost: recontextualizare, nu distrugere."""
    
    def __init__(self, max_age_days=72):
        self.items = []
        self.compost = []  # [FIX] era folosit fără să fie inițializat
        self.max_age_days = max_age_days

    def add(self, memory):
        memory["expires_at"] = datetime.now() + timedelta(days=self.max_age_days)
        self.items.append(memory)

    def process(self):
        now = datetime.now()
        for item in self.items[:]:
            if now > item["expires_at"]:
                # Extrage tiparul înainte de compostare
                pattern = self.extract_pattern(item)
                self.compost.append(pattern)
                self.items.remove(item)

    def extract_pattern(self, item):
        return {
            "type": "composted_pattern",
            "origin": item.get("id"),
            "summary": item.get("summary", ""),
            "composted_at": datetime.now().isoformat()
        }
